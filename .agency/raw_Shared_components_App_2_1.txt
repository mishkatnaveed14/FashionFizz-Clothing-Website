import React, { useState, useEffect } from 'react';
import { useNavigate } from 'react-router-dom';
import { useStore } from '../lib/store';
import { motion, AnimatePresence } from 'framer-motion';
import { X, ShoppingBag, Trash2, Plus, Minus, ArrowRight, Tag } from 'lucide-react';
import { formatCurrency } from '../lib/data';

export default function CartDrawer() {
  const { cart, removeFromCart, updateCartQuantity, discounts, notify } = useStore();
  const [isOpen, setIsOpen] = useState(false);
  const [promoCode, setPromoCode] = useState('');
  const [appliedDiscount, setAppliedDiscount] = useState(null);
  const navigate = useNavigate();

  useEffect(() => {
    const handleToggle = () => setIsOpen((prev) => !prev);
    window.addEventListener('toggle-cart', handleToggle);
    return () => window.removeEventListener('toggle-cart', handleToggle);
  }, []);

  useEffect(() => {
    if (isOpen) {
      document.body.style.overflow = 'hidden';
    } else {
      document.body.style.overflow = 'unset';
    }
    return () => { document.body.style.overflow = 'unset'; };
  }, [isOpen]);

  const subtotal = cart.reduce((acc, item) => {
    const price = item.product.sale_price || item.product.price;
    return acc + price * item.quantity;
  }, 0);

  const discountAmount = appliedDiscount ? (subtotal * appliedDiscount.discount_percentage) / 100 : 0;
  const total = subtotal - discountAmount;

  const handleApplyPromo = (e) => {
    e.preventDefault();
    const found = discounts.find((d) => d.code.toUpperCase() === promoCode.toUpperCase() && d.status === 'Active');
    if (found) {
      setAppliedDiscount(found);
      notify(`Promo code ${found.code} applied! ${found.discount_percentage}% off`);
    } else {
      notify('Invalid or expired promo code', 'error');
    }
  };

  const handleCheckout = () => {
    setIsOpen(false);
    navigate('/checkout');
  };

  return (
    <AnimatePresence>
      {isOpen && (
        <div className="fixed inset-0 z-50 flex justify-end">
          <motion.div
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            onClick={() => setIsOpen(false)}
            className="fixed inset-0 bg-paper/80 backdrop-blur-md"
          />
          <motion.div
            initial={{ x: '100%' }}
            animate={{ x: 0 }}
            exit={{ x: '100%' }}
            transition={{ type: 'tween', duration: 0.3, ease: [0.16, 1, 0.3, 1] }}
            className="relative bg-paper-surface border-l border-line w-full max-w-md h-full flex flex-col z-10"
          >
            {/* Header */}
            <div className="flex items-center justify-between p-6 border-b border-line">
              <div className="flex items-center gap-3">
                <ShoppingBag size={20} className="text-brand" />
                <h3 className="font-display font-semibold text-xl text-ink">Shopping Bag ({cart.length})</h3>
              </div>
              <button
                onClick={() => setIsOpen(false)}
                className="text-ink-muted hover:text-ink p-2 transition-colors cursor-pointer"
              >
                <X size={20} />
              </button>
            </div>

            {/* Items List */}
            <div className="flex-1 overflow-y-auto p-6 flex flex-col gap-4">
              {cart.length === 0 ? (
                <div className="h-full flex flex-col items-center justify-center text-center gap-4">
                  <ShoppingBag size={48} className="text-line" />
                  <p className="font-display text-lg text-ink-muted">Your shopping bag is empty</p>
                  <button
                    onClick={() => setIsOpen(false)}
                    className="bg-brand text-paper font-mono text-xs uppercase font-bold py-3 px-6 rounded-[2px]"
                  >
                    Explore Collections
                  </button>
                </div>
              ) : (
                cart.map((item) => {
                  const itemKey = `${item.product.id}-${item.size}-${item.color}`;
                  const price = item.product.sale_price || item.product.price;
                  return (
                    <div key={itemKey} className="flex gap-4 p-4 bg-paper border border-line rounded-[4px] relative">
                      <div className="w-20 h-24 bg-soft border border-line rounded-[2px] flex items-center justify-center shrink-0">
                        <span className="font-display text-xs text-ink-muted">FF</span>
                      </div>
                      <div className="flex flex-col flex-1 justify-between">
                        <div className="flex justify-between items-start gap-2">
                          <h4 className="font-body font-medium text-sm text-ink line-clamp-1">{item.product.title}</h4>
                          <button
                            onClick={() => removeFromCart(itemKey)}
                            className="text-ink-muted hover:text-red-400 transition-colors"
                          >
                            <Trash2 size={16} />
                          </button>
                        </div>
                        <div className="flex items-center gap-2 font-mono text-xs text-ink-muted">
                          <span>Size: {item.size}</span>
                          <span>•</span>
                          <span>Color: {item.color}</span>
                        </div>
                        <div className="flex items-center justify-between mt-2">
                          <span className="font-mono text-sm font-bold text-brand">{formatCurrency(price)}</span>
                          <div className="flex items-center border border-line rounded-[2px] bg-paper-surface">
                            <button
                              onClick={() => updateCartQuantity(itemKey, item.quantity - 1)}
                              className="px-2.5 py-1 text-ink-muted hover:text-ink"
                            >
                              <Minus size={12} />
                            </button>
                            <span className="font-mono text-xs px-2 text-ink">{item.quantity}</span>
                            <button
                              onClick={() => updateCartQuantity(itemKey, item.quantity + 1)}
                              className="px-2.5 py-1 text-ink-muted hover:text-ink"
                            >
                              <Plus size={12} />
                            </button>
                          </div>
                        </div>
                      </div>
                    </div>
                  );
                })
              )}
            </div>

            {/* Footer / Summary */}
            {cart.length > 0 && (
              <div className="p-6 border-t border-line bg-paper flex flex-col gap-4">
                {/* Promo Code Form */}
                <form onSubmit={handleApplyPromo} className="flex gap-2">
                  <div className="relative flex-1">
                    <Tag size={14} className="absolute left-3 top-1/2 -translate-y-1/2 text-ink-muted" />

                      type="text"
                      value={promoCode}
                      onChange={(e) => setPromoCode(e.target.value)}
                      placeholder="Promo Code (e.g. AURA50)"
                      className="w-full bg-paper-surface border border-line text-ink pl-9 pr-3 py-2.5 rounded-[2px] font-mono text-xs uppercase outline-none focus:border-brand"
                    />

                  <button
                    type="submit"
                    className="bg-soft border border-line text-ink font-mono text-xs uppercase px-4 rounded-[2px] hover:border-brand"
                  >
                    Apply

                </form>

                {appliedDiscount && (
                  <div className="flex items-center justify-between font-mono text-xs text-emerald-400 bg-emerald-500/10 p-2 rounded-[2px]">
                    <span>Discount Code: {appliedDiscount.code} ({appliedDiscount.discount_percentage}% OFF)</span>
                    <button onClick={() => setAppliedDiscount(null)} className="hover:text-white">Remove</button>
                  </div>
                )}

                <div className="flex flex-col gap-2 font-mono text-xs text-ink-muted pt-2 border-t border-line">

                    <span>{formatCurrency(subtotal)}</span>

        <div className="fixed inset-0 z-50 flex items-start justify-center pt-20 p-4 bg-paper/90 backdrop-blur-md">
          <motion.div
            initial={{ opacity: 0, scale: 0.95, y: -20 }}
            animate={{ opacity: 1, scale: 1, y: 0 }}
            exit={{ opacity: 0, scale: 0.95, y: -20 }}
            className="bg-paper-surface border border-line w-full max-w-2xl rounded-[4px] shadow-[0_24px_64px_rgba(0,0,0,0.8)] overflow-hidden flex flex-col"
          >
            {/* Search Input Bar */}
            <div className="flex items-center px-6 py-4 border-b border-line gap-4">
              <Search size={20} className="text-brand shrink-0" />
              <input
                type="text"
                value={query}
                onChange={(e) => setQuery(e.target.value)}
                placeholder="Search unstitched, pret, festive wear, SKU..."
                autoFocus
                className="w-full bg-transparent text-ink font-body text-lg outline-none"
              />
              <button
                onClick={() => setIsOpen(false)}
                className="text-ink-muted hover:text-ink p-2 transition-colors cursor-pointer"
              >
                <X size={20} />
              </button>
            </div>

            {/* Quick Category Tags */}
            <div className="px-6 py-3 bg-paper border-b border-line flex items-center gap-2 overflow-x-auto">
              <span className="font-mono text-[10px] uppercase text-ink-muted shrink-0">Popular:</span>
              {['Unstitched', 'Pret', 'Velvet', 'Lawn', 'Fragrance', 'Men'].map((tag) => (
                <button
                  key={tag}
                  onClick={() => setQuery(tag)}
                  className="bg-paper-surface border border-line px-3 py-1 rounded-full font-mono text-[10px] text-ink hover:border-brand shrink-0"
                >
                  {tag}
                </button>
              ))}
            </div>

            {/* Results */}
            <div className="max-h-[60vh] overflow-y-auto p-6 flex flex-col gap-4">
              {query.trim() !== '' && filteredProducts.length === 0 ? (
                <div className="py-12 text-center">
                  <p className="font-mono text-sm text-ink-muted">No luxury pieces found matching "{query}"</p>
                </div>
              ) : query.trim() === '' ? (
                <div className="py-12 text-center">
                  <p className="font-mono text-xs uppercase tracking-widest text-ink-muted">Type to search the Aura Chromatique atelier</p>
                </div>
              ) : (
                filteredProducts.map((prod) => (
                  <Link
                    key={prod.id}
                    to={`/product/${prod.id}`}
                    onClick={() => setIsOpen(false)}
                    className="flex items-center justify-between p-4 bg-paper border border-line rounded-[4px] hover:border-brand transition-all group"
                  >
                    <div className="flex items-center gap-4">
                      <div className="w-12 h-14 bg-soft border border-line rounded-[2px] flex items-center justify-center shrink-0">
                        <span className="font-display text-xs text-ink-muted">FF</span>
                      </div>
                      <div className="flex flex-col gap-1">
                        <span className="font-mono text-[10px] uppercase text-brand tracking-widest">{prod.category} / {prod.sub_category}</span>
                        <h4 className="font-body font-medium text-sm text-ink group-hover:text-brand transition-colors">{prod.title}</h4>
                        <span className="font-mono text-xs text-ink-muted">{formatCurrency(prod.sale_price || prod.price)}</span>
                      </div>
                    </div>
                    <ArrowRight size={18} className="text-ink-muted group-hover:text-brand transition-colors" />
                  </Link>
                ))
              )}
            </div>
          </motion.div>
        </div>
      )}
    </AnimatePresence>
  );
}
